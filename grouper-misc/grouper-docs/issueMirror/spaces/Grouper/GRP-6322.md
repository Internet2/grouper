# GRP-6322  add ai sql query for added to group in last week

{code:java}
select count(1)
from grouper_pit_memberships_lw_v gpmlv
join grouper_pit_groups gpg on gpg.id = gpmlv.owner_group_id
join grouper_pit_fields gpf on gpf.id = gpmlv.field_id
join grouper_pit_members gpm on gpm.id = gpmlv.member_id
join grouper_members gm on gm.id = gpm.source_id
where gpg.name = 'group_name'
and gm.subject_source = 'subjectSourceId'
and gpf.name = 'members'
and grouper_to_timestamp(gpmlv.the_start_time) >= current_timestamp - interval '7 days';
 {code}
