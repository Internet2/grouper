package edu.upenn.isc.fastGrouper;

import org.quartz.Scheduler;
import org.quartz.SchedulerException;

import edu.internet2.middleware.grouper.app.loader.GrouperLoader;
import edu.upenn.isc.fast.daemon.DaemonStoppedException;
import edu.upenn.isc.fast.daemon.DaemonTask;
import edu.upenn.isc.fast.log.FastLogger;
import edu.upenn.isc.fast.util.FastThreadUtils;


/**
 * sync up the view that goes against penn community with a flat
 * table so that searches and lookups are super fast.
 */
public class GrouperLoaderDaemon extends DaemonTask {

  /**
   * 
   */
  public GrouperLoaderDaemon() {
  }

  /**
   * @param runStandalone
   */
  public GrouperLoaderDaemon(boolean runStandalone) {
    super(runStandalone);
  }

  /**
   * @param runStandalone
   * @param name
   */
  public GrouperLoaderDaemon(boolean runStandalone, String name) {
    super(runStandalone, name);
  }

  /** logger */
  private static final FastLogger log = new FastLogger(GrouperLoaderDaemon.class);
  
  /**
   * @see edu.upenn.isc.fast.daemon.DaemonTask#run()
   */
  @Override
  public boolean run() {
    
    try {
      //kick off the loader
      GrouperLoader.main(null);
      
      //now sleep forever, checking the scheduler periodically
      while(true) {
        
        //safe-sleep a minute
        this.sleep(1000*60);
        try {
          Scheduler scheduler = GrouperLoader.schedulerFactory().getScheduler();
          if (scheduler.isShutdown()) {
            return false;
          }
        } catch (SchedulerException se) {
          throw new RuntimeException("Problem checking scheduler!", se);
        }
      }
    } catch (DaemonStoppedException dse) {
      //this happens if someone stops the daemon
      try {
        //stop all jobs and exit
        final Scheduler scheduler = GrouperLoader.schedulerFactory().getScheduler();
        //dont start any new jobs
        scheduler.standby();
        //wait for 9 seconds
        FastThreadUtils.sleep(9000);
        //shut it down and kill any jobs...
        scheduler.shutdown();
      } catch (SchedulerException se) {
        //dont rethrow, causes the daemon to retry
        log.error("Problem shutting down the scheduler!", se);
      }
      throw dse;
    }
  }

}
