CREATE TABLE penngrouper.penn_box_user (
 subject_id varchar(100) NULL,
 box_id varchar(20) NOT NULL,
 created_at timestamp NULL,
 login varchar(100) NULL,
 modified_at timestamp NULL,
 "name" varchar(100) NULL,
 space_used int8 NULL,
 status varchar(40) NULL,
 CONSTRAINT penn_box_user_pk PRIMARY KEY (box_id)
);
CREATE UNIQUE INDEX penn_box_user_login_idx ON penngrouper.penn_box_user USING btree (login);
CREATE UNIQUE INDEX penn_box_user_subject_id_idx ON penngrouper.penn_box_user USING btree (subject_id); {code}
