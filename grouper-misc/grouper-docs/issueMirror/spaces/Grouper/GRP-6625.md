### Chris Hyzer - 2026-02-20T18:16:20.277+0000

find folders or groups with partially assigned types

 
{code:java}
select stem_name as object_name, 'stem' as type
from grouper_attr_asn_stem_v gaasv
where gaasv.attribute_def_name_name = 'etc:objectTypes:grouperObjectTypeMarker'
and not exists (
  select 1 from grouper_aval_asn_asn_stem_v gaaasv
  where gaaasv.stem_name = gaasv.stem_name
  and gaaasv.attribute_def_name_name1 = 'etc:objectTypes:grouperObjectTypeMarker'
  and gaaasv.attribute_def_name_name2 = 'etc:objectTypes:grouperObjectTypeName'
  and gaaasv.value_string is not null
  and gaaasv.attribute_assign_id1 = gaasv.attribute_assign_id
)
union all
select group_name as object_name, 'group' as type
from grouper_attr_asn_group_v gaagv
where gaagv.attribute_def_name_name = 'etc:objectTypes:grouperObjectTypeMarker'
and not exists (
  select 1 from grouper_aval_asn_asn_group_v gaaagv
  where gaaagv.group_name = gaagv.group_name
  and gaaagv.attribute_def_name_name1 = 'etc:objectTypes:grouperObjectTypeMarker'
  and gaaagv.attribute_def_name_name2 = 'etc:objectTypes:grouperObjectTypeName'
  and gaaagv.value_string is not null
  and gaaagv.attribute_assign_id1 = gaagv.attribute_assign_id
)
