# GRP-221  add a field view

Add a field view, e.g.

select gt.name as type_name, gf.name as field_name, gf.is_nullable, gf.type as field_type, 
gf.read_privilege, 
gf.write_privilege, gf.id as field_id, gt.id as type_id, gt.is_assignable, gt.is_internal
  from grouper_fields gf, grouper_types gt where 
gf.grouptype_uuid = gt.id
