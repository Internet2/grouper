# GRP-7076  Group names of 256–1024 chars fail to save with an opaque DB error: grouper_members.subject_identifier0 is VARCHAR(255) but grouper_groups.name is VARCHAR(1024)

GRP-7076 widens these columns from {{varchar(255)}} to {{{}varchar(1024){}}}:
 * {{{}grouper_members.subject_identifier0{}}}, {{{}subject_identifier1{}}}, {{subject_identifier2}}
 * {{grouper_pit_members.subject_identifier0}}
 * {{{}grouper_stems.name{}}}, {{{}display_name{}}}, {{alternate_name}}

Oracle/MySQL are handled automatically by {{{}UpgradeTaskV43{}}}. PostgreSQL cannot be: these columns are referenced by views, and PostgreSQL refuses {{ALTER COLUMN ... TYPE}} on a view-referenced column. The task logs {{{}GRP-7076 MANUAL ACTION REQUIRED{}}}, records itself done (so the upgrade isn't blocked), and leaves the widening to this runbook. Run as the schema owner; the queries discover objects in _your_ DB (nothing hard-coded).  Note: if you are logically replicating any of these columns you need to adjust them in the target database(s).  Note: if you do not make these changes then you will only notice if you try to make a group or stem with a long name.
h2. The dependency CTE (prepend to every 0a-0e query)

Finds the *transitive closure* of dependent views (a view depending on another dependent view is included), so the capture is complete.

 
{code:java}
WITH RECURSIVE base AS (
  SELECT DISTINCT dependee.oid AS view_oid
  FROM pg_depend d
  JOIN pg_rewrite r        ON r.oid = d.objid
  JOIN pg_class   dependee ON dependee.oid = r.ev_class
  JOIN pg_class   dependent ON dependent.oid = d.refobjid
  JOIN pg_attribute a      ON a.attrelid = d.refobjid AND a.attnum = d.refobjsubid
  WHERE dependee.relkind = 'v'
    AND ( (dependent.relname = 'grouper_members'     AND a.attname IN ('subject_identifier0','subject_identifier1','subject_identifier2'))
       OR (dependent.relname = 'grouper_pit_members' AND a.attname = 'subject_identifier0')
       OR (dependent.relname = 'grouper_stems'       AND a.attname IN ('name','display_name','alternate_name')) )
),
closure AS (
  SELECT view_oid FROM base
  UNION
  SELECT dependee.oid
  FROM closure c
  JOIN pg_depend  d        ON d.refobjid = c.view_oid
  JOIN pg_rewrite r        ON r.oid = d.objid
  JOIN pg_class   dependee ON dependee.oid = r.ev_class AND dependee.relkind = 'v'
  WHERE dependee.oid <> c.view_oid
) {code}
 
h2. Step 0 - capture (changes nothing; save each result)

*0a. List dependent views*
 
{code:java}
-- <CTE>
SELECT DISTINCT (view_oid::regclass)::text AS view_name FROM closure ORDER BY 1; {code}

 
*0b. View source*
 
{code:java}
-- <CTE>
SELECT 'CREATE VIEW ' || (view_oid::regclass)::text || ' AS ' || pg_get_viewdef(view_oid, true)
FROM closure ORDER BY 1; {code}

 
*0c. View + column comments*
 
 
{code:java}
-- <CTE>
SELECT 'COMMENT ON VIEW ' || (view_oid::regclass)::text || ' IS ' || quote_literal(obj_description(view_oid,'pg_class')) || ';'
FROM closure WHERE obj_description(view_oid,'pg_class') IS NOT NULL
UNION ALL
SELECT 'COMMENT ON COLUMN ' || (c.view_oid::regclass)::text || '.' || quote_ident(a.attname) || ' IS ' || quote_literal(col_description(c.view_oid,a.attnum)) || ';'
FROM closure c JOIN pg_attribute a ON a.attrelid=c.view_oid AND a.attnum>0 AND NOT a.attisdropped
WHERE col_description(c.view_oid,a.attnum) IS NOT NULL ORDER BY 1; {code}

 
*0d. Grants + ownership* (site-specific - Grouper does NOT restore these)
{code:java}
-- <CTE>
SELECT 'ALTER VIEW ' || (view_oid::regclass)::text || ' OWNER TO ' || quote_ident(pg_get_userbyid((SELECT relowner FROM pg_class WHERE oid=view_oid))) || ';'
FROM closure
UNION ALL
SELECT 'GRANT ' || g.privilege_type || ' ON ' || (c.view_oid::regclass)::text || ' TO '
       || CASE WHEN g.grantee=0 THEN 'PUBLIC' ELSE quote_ident(rr.rolname) END
       || CASE WHEN g.is_grantable THEN ' WITH GRANT OPTION' ELSE '' END || ';'
FROM closure c JOIN pg_class pc ON pc.oid=c.view_oid
CROSS JOIN LATERAL aclexplode(pc.relacl) g
LEFT JOIN pg_roles rr ON rr.oid=g.grantee ORDER BY 1; {code}
 
*0e. DROP statements*
 
 
{code:java}
-- <CTE>
SELECT 'DROP VIEW IF EXISTS ' || (view_oid::regclass)::text || ' CASCADE;' FROM closure ORDER BY 1; {code}

Run the {{DROP ... CASCADE}} from 0e.

h2. Step 3 - recreate views/comments/grants
 * *Option A (recommended):* run a Grouper registry DDL pass (or start with auto-DDL) → the engine recreates all standard views + comments; then replay only the *0d* grants/ownership.
 * *Option B (fully manual):* replay 0b → 0c → 0d; if a CREATE VIEW fails on a not-yet-created dependency, re-run the 0b block (2-3 passes resolve the nesting).

h2. Step 4 - verify
{code:java}
SELECT table_name, column_name, character_maximum_length FROM information_schema.columns
WHERE (table_name='grouper_members'     AND column_name IN ('subject_identifier0','subject_identifier1','subject_identifier2'))
   OR (table_name='grouper_pit_members' AND column_name='subject_identifier0')
   OR (table_name='grouper_stems'       AND column_name IN ('name','display_name','alternate_name'))
ORDER BY 1,2;  -- all should be 1024
 {code}
