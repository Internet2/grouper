# GRP-2698  add setting for email to be sent to one address for non prod

grouper.properties
{code:java}
# in non-prod send all messages to this address to see what messages would have been sent out (or comma separate multiple addresses).
# if you set addresses here, no real email will be sent out
# {valueType: "string"}
#mail.sendAllMessagesHere = a@b.c {code}
Here is an example of a message sent to the single address:
{code:java}
TO: whomeverWouldHaveReceived@example.com
CC: whomeverWouldHaveBeenCced@example.com
BODY:

hey there {code}
