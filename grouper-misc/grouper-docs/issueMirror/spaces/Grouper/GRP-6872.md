###########################################################################
## ABAC (Attribute Based Access Control)
###########################################################################

# Comma-separated list of subject source IDs to include in ABAC script evaluation.
# If blank, all non-internal non-grouper subject sources will be used (default behavior).
# Imagine you had 100 ABAC groups, which subject source or sources would the majority have?
# That should be your default(s).  Generally this should be your one person subject source.
# e.g. univPerson
# {valueType: "string"}
grouper.abac.globalDefaultSubjectSourceIds =

# Whether users can override the default subject source IDs on a per-group basis.
# If false, the per-group attribute is ignored and the UI never shows the source picker in read/write mode.
# Unless you have a specific need, this should be false to avoid confusion and unexpected results.
# {valueType: "boolean"}
grouper.abac.allowUserOverrideSubjectSourceIds = false

# Comma-separated list of subject source IDs available for users to pick from when overriding defaults.
# If blank, no UI picker is shown and per-group config is ignored even if 
# grouper.abac.allowUserOverrideSubjectSourceIds is true.
# This should be a superset of or equal to grouper.abac.globalDefaultSubjectSourceIds.
# e.g. univPerson, grouperEntities
# {valueType: "string"}
grouper.abac.availableSubjectSourceIds =
