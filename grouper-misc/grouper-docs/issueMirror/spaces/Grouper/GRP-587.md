# GRP-587  [ldappcng] should support object renaming

ldappcng should support object renaming.

https://lists.internet2.edu/sympa/arc/grouper-users/2011-03/msg00064.html

## Comments

### tzeller@example.com - 2011-11-10T00:32:15.748+0000

Objects are renameable given an <alternateIdentifier /> element in the ldappcng configuration which refers to an attribute definition whose value is a PSOIdentifier. The <alternateIdentifier /> is the "old" identifier, while the <identifier /> element is the "new" identifier.

The <alternateIdentifier /> element is optional, and if not present, renames will not be attempted.

For example :

Ldappcng configuration :

<object id="group" authoritative="true">
  <identifier ref="groupDn" baseId="ou=testgroups,${base}">
  <alternateIdentifier ref="groupDnAlternate" />
  <alternateIdentifier ref="groupDnAlternateChangeLog" />

Attribute resolver configuration :

  <!-- The LDAP DN of a group. For example, "cn=groupExtension,ou=stem,ou=testgroups,dc=edu". -->
  <resolver:AttributeDefinition id="groupDn" xsi:type="ldappc:LdapDnPSOIdentifier"
    structure="bushy" sourceAttributeID="name" rdnAttributeName="cn" base="ou=testgroups,${base}">
    <!-- Dependencies which return a "name" attribute whose value is the group name. -->
    <resolver:Dependency ref="GroupDataConnector" />
    <resolver:Dependency ref="DeleteGroupChangeLogDataConnector" />
    <resolver:Dependency ref="UpdateGroupChangeLogDataConnector" />
  </resolver:AttributeDefinition>

  <!-- The alternate LDAP DN of a group. For example, the DN of a group before it is renamed. -->
  <resolver:AttributeDefinition id="groupDnAlternate" xsi:type="ldappc:LdapDnPSOIdentifier"
    structure="bushy" sourceAttributeID="alternateName" rdnAttributeName="cn" base="ou=testgroups,${base}">
    <!-- Dependency which returns an "alternateName" attribute whose value is the old group name. -->
    <resolver:Dependency ref="GroupDataConnector" />
  </resolver:AttributeDefinition>

  <!-- The alternate LDAP DN of a group via the change log. For example, the DN of a group before it is renamed. -->
  <resolver:AttributeDefinition id="groupDnAlternateChangeLog" xsi:type="ldappc:LdapDnPSOIdentifier"
    structure="bushy" sourceAttributeID="propertyOldValue" rdnAttributeName="cn" base="ou=testgroups,${base}">
    <!-- Dependency which returns a "propertyOldValue" attribute whose value is the old group name. -->
    <resolver:Dependency ref="UpdateGroupChangeLogDataConnector" />
  </resolver:AttributeDefinition>

  <!-- Returns change log attributes representing the changing of a group's name. -->
  <resolver:DataConnector id="UpdateGroupChangeLogDataConnector" xsi:type="grouper:ChangeLogDataConnector">
    <!-- The AND filter matches both child filters. -->
    <grouper:Filter xsi:type="grouper:AND">
      <!-- The ChangeLogEntry filter matches change log entries with the given category and action. -->
      <grouper:Filter xsi:type="grouper:ChangeLogEntry" category="group" action="updateGroup" />
      <!-- The ChangeLogExactAttribute filter matches change log entries with the given attribute and value. -->
      <grouper:Filter xsi:type="grouper:ChangeLogExactAttribute" name="propertyChanged" value="name" />
    </grouper:Filter>
  </resolver:DataConnector>
