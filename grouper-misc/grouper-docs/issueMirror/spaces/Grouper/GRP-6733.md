# GRP-6733  add configs to allow various types of authentication for mcp

oauth defaults to true, others default to false
{code:java}
# Allow OAuth JWT Bearer authentication for MCP.
# {valueType: "boolean", defaultValue: "true"}
grouper.mcp.auth.oauth = true

# Allow HTTP Basic authentication for MCP.
# Requires grouper.is.ws.basicAuthn = true in grouper.hibernate.properties.
# {valueType: "boolean", defaultValue: "false"}
grouper.mcp.auth.httpBasic = false

# Allow container-managed authentication for MCP (e.g. Tomcat valve, Apache module).
# {valueType: "boolean", defaultValue: "false"}
grouper.mcp.auth.container = false

# Allow custom authentication class for MCP (ws.security.non-rampart.authentication.class).
# {valueType: "boolean", defaultValue: "false"}
grouper.mcp.auth.customAuthClass = false


 {code}
