// GSH script to check data field configuration referential integrity
// Paste this into Grouper Shell (GSH) to check for broken references

import java.util.*;
import org.apache.commons.lang3.*;
import edu.internet2.middleware.grouper.*;
import edu.internet2.middleware.grouper.cfg.*;
import edu.internet2.middleware.grouper.dataField.*;
import edu.internet2.middleware.grouper.misc.*;
import edu.internet2.middleware.grouper.util.*;

GrouperDataEngine grouperDataEngine = new GrouperDataEngine();
grouperDataEngine.loadFieldsAndRows(null);

Map fieldConfigMap = grouperDataEngine.getFieldConfigByConfigId();
Map rowConfigMap = grouperDataEngine.getRowConfigByConfigId();
Map privacyRealmMap = grouperDataEngine.getPrivacyRealmConfigByConfigId();

int errorCount = 0;

// Check 1: Each data field's privacy realm reference must exist
System.out.println("=== Checking data field -> privacy realm references ===");
Iterator fieldEntries = fieldConfigMap.entrySet().iterator();
while (fieldEntries.hasNext()) {
  Map.Entry fieldEntry = (Map.Entry) fieldEntries.next();
  String fieldConfigId = (String) fieldEntry.getKey();
  GrouperDataFieldConfig fieldConfig = (GrouperDataFieldConfig) fieldEntry.getValue();
  String privacyRealmId = fieldConfig.getGrouperPrivacyRealmConfigId();
  if (StringUtils.isNotBlank(privacyRealmId) && !privacyRealmMap.containsKey(privacyRealmId)) {
    System.out.println("ERROR: Data field '" + fieldConfigId + "' references privacy realm '" + privacyRealmId + "' which does not exist.  Property: grouperDataField." + fieldConfigId + ".fieldPrivacyRealm");
    errorCount++;
  }
}

// Check 2: Each data row's field references must exist
System.out.println("=== Checking data row -> data field references ===");
Iterator rowEntries = rowConfigMap.entrySet().iterator();
while (rowEntries.hasNext()) {
  Map.Entry rowEntry = (Map.Entry) rowEntries.next();
  String rowConfigId = (String) rowEntry.getKey();
  GrouperDataRowConfig rowConfig = (GrouperDataRowConfig) rowEntry.getValue();
  Set dataFieldConfigIds = rowConfig.getDataFieldConfigIds();
  Iterator fieldIdIter = dataFieldConfigIds.iterator();
  while (fieldIdIter.hasNext()) {
    String dataFieldConfigId = (String) fieldIdIter.next();
    if (!fieldConfigMap.containsKey(dataFieldConfigId)) {
      System.out.println("ERROR: Data row '" + rowConfigId + "' references data field '" + dataFieldConfigId + "' which does not exist.  Property: grouperDataRow." + rowConfigId + ".rowDataField.*.colDataFieldConfigId");
      errorCount++;
    }
  }
}

// Check 3: Each data row's privacy realm reference must exist
System.out.println("=== Checking data row -> privacy realm references ===");
rowEntries = rowConfigMap.entrySet().iterator();
while (rowEntries.hasNext()) {
  Map.Entry rowEntry = (Map.Entry) rowEntries.next();
  String rowConfigId = (String) rowEntry.getKey();
  GrouperDataRowConfig rowConfig = (GrouperDataRowConfig) rowEntry.getValue();
  String privacyRealmId = rowConfig.getPrivacyRealmName();
  if (StringUtils.isNotBlank(privacyRealmId) && !privacyRealmMap.containsKey(privacyRealmId)) {
    System.out.println("ERROR: Data row '" + rowConfigId + "' references privacy realm '" + privacyRealmId + "' which does not exist.  Property: grouperDataRow." + rowConfigId + ".rowPrivacyRealm");
    errorCount++;
  }
}

// Check 4: Privacy realm group references must point to existing groups
System.out.println("=== Checking privacy realm -> group references ===");
GrouperSession grouperSession = GrouperSession.startRootSession();
Iterator realmEntries = privacyRealmMap.entrySet().iterator();
while (realmEntries.hasNext()) {
  Map.Entry realmEntry = (Map.Entry) realmEntries.next();
  String realmConfigId = (String) realmEntry.getKey();
  GrouperPrivacyRealmConfig realmConfig = (GrouperPrivacyRealmConfig) realmEntry.getValue();
  String viewersGroupName = realmConfig.getPrivacyRealmViewersGroupName();
  if (StringUtils.isNotBlank(viewersGroupName)) {
    Group viewersGroup = GroupFinder.findByName(viewersGroupName, false);
    if (viewersGroup == null) {
      System.out.println("ERROR: Privacy realm '" + realmConfigId + "' viewers group '" + viewersGroupName + "' does not exist.  Property: grouperPrivacyRealm." + realmConfigId + ".privacyRealmViewersGroupName");
      errorCount++;
    }
  }
  String updatersGroupName = realmConfig.getPrivacyRealmUpdatersGroupName();
  if (StringUtils.isNotBlank(updatersGroupName)) {
    Group updatersGroup = GroupFinder.findByName(updatersGroupName, false);
    if (updatersGroup == null) {
      System.out.println("ERROR: Privacy realm '" + realmConfigId + "' updaters group '" + updatersGroupName + "' does not exist.  Property: grouperPrivacyRealm." + realmConfigId + ".privacyRealmUpdatersGroupName");
      errorCount++;
    }
  }
  String readersGroupName = realmConfig.getPrivacyRealmReadersGroupName();
  if (StringUtils.isNotBlank(readersGroupName)) {
    Group readersGroup = GroupFinder.findByName(readersGroupName, false);
    if (readersGroup == null) {
      System.out.println("ERROR: Privacy realm '" + realmConfigId + "' readers group '" + readersGroupName + "' does not exist.  Property: grouperPrivacyRealm." + realmConfigId + ".privacyRealmReadersGroupName");
      errorCount++;
    }
  }
}
GrouperSession.stopQuietly(grouperSession);

// Summary
System.out.println("=== Summary ===");
System.out.println("Privacy realms: " + privacyRealmMap.size());
System.out.println("Data fields: " + fieldConfigMap.size());
System.out.println("Data rows: " + rowConfigMap.size());
if (errorCount == 0) {
  System.out.println("Result: All referential integrity checks PASSED");
} else {
  System.out.println("Result: Found " + errorCount + " referential integrity error(s)");
