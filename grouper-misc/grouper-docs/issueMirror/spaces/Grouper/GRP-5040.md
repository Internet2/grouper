# GRP-5040  add option to not have ability to delete built in attributes (defs and names)

like this

# if this is true, attribute definitions can only be deleted via gsh or WS, to prevent accidental deletion, including cascading deletion of all attribute names and their assignments
# {valueType: "boolean", required: true}
uiV2.attributeDef.preventDeleteBuiltInInUi = false

# if this is true, attribute names can only be deleted via gsh or WS, to prevent accidental deletion, including cascading deletion of all attribute assignments
# {valueType: "boolean", required: true}
uiV2.attributeDefName.preventDeleteBuiltInInUi = false


groups and folders
