#########
# Given a group containing another group, return source and subjectIds of members of the larger group who will be removed if the smaller group is removed
# https://spaces.at.internet2.edu/x/UAj5AQ
#########

select gm.subject_source, gm.subject_id
from grouper_members gm
where exists (select 1 from grouper_memberships_lw_v gmlv
              -- group to remove
              where gmlv.member_id = gm.id and gmlv.group_name = 'test:testGroup3'
              and gmlv.list_name = 'members')
and 1 = (select count(1) from grouper_memberships_all_v gmav, grouper_fields gf, grouper_groups gg
              -- overall group
              where gmav.member_id = gm.id and gmav.owner_group_id = gg.id and gg.name = 'test:testGroup5'
              and gmav.field_id = gf.id and gf.name = 'members')
and not exists (select 1 from grouper_memberships_all_v gmav, grouper_fields gf, grouper_groups gg
              -- overall group
              where gmav.member_id = gm.id and gmav.owner_group_id = gg.id and gg.name = 'test:testGroup5'
              and gmav.field_id = gf.id and gf.name = 'members' and gmav.mship_type = 'immediate')

# Make a SQL that generates GSH
# https://spaces.at.internet2.edu/x/UAj5AQ
#######

select distinct attribute_assign_id_marker, assign_name,
'new Provisionable' ||
  (case when exists (select 1 from temp_provisioning_assigns tpa2 where tpa.attribute_assign_id_marker = tpa2.attribute_assign_id_marker
  and tpa2.assign_type = 'group') then
    'Group' else 'Stem' end)
  || 'Save().assignTargetName("'
  || (select tpa2.value_string from temp_provisioning_assigns tpa2 where tpa.attribute_assign_id_marker = tpa2.attribute_assign_id_marker
and tpa2.attribute_name = 'etc:provisioning:provisioningTarget')
  || '").assign' ||
  (case when exists (select 1 from temp_provisioning_assigns tpa2 where tpa.attribute_assign_id_marker = tpa2.attribute_assign_id_marker
  and tpa2.assign_type = 'group') then
    'Group' else 'Stem' end) ||
  'Name("' ||
  (select distinct assign_name from temp_provisioning_assigns tpa2 where tpa.attribute_assign_id_marker = tpa2.attribute_assign_id_marker) ||
   '")"' ||
   (case when exists (select 1 from temp_provisioning_assigns tpa2 where tpa.attribute_assign_id_marker = tpa2.attribute_assign_id_marker
  and tpa2.assign_type = 'group') then
    'Group' else 'Stem' end)
   || '".save();'
as script from temp_provisioning_assigns tpa
where exists (select 1 from temp_provisioning_assigns tpa2
where tpa.attribute_assign_id_marker = tpa2.attribute_assign_id_marker and tpa2.value_string = 'true'
and tpa2.attribute_name = 'etc:provisioning:provisioningDirectAssign')
and assign_type = 'stem'
order by 2;

SELECT subject_id,
(select grouper_to_timestamp(min(gpmglv.the_start_time)) FROM grouper_pit_mship_group_lw_v gpmglv
where gpmglv.group_name = gmlv.group_name
and gpmglv.member_id = gmlv.member_id
and gpmglv.field_name = 'members'
and gpmglv.the_end_time is null
)
FROM grouper_memberships_lw_v gmlv
WHERE group_name = 'test:testGroup' AND list_name = 'members' AND subject_source = 'schoolPerson';


#######
# SQL to find all past and present memberships for a grouper_members.subject_identifier0 value.
#... (review results)
#convert start_time and end_time to human readable dates and times.  ( Which it got wrong.. only / 1000 instead of /1000000 )
#... (review results)
#change order so that highest values of pitms.start_time and pitms.end_time are at the top of the results
######

select *
from(
    select
        gm.subject_id,
        gm.subject_identifier0,
        gg.name as group_name,
        gf.name as field_name,
        gmship.enabled,
        to_timestamp(gmship.create_time / 1000000) as start_time,
        null as end_time,
        gg.display_name
        ,to_timestamp(gmship.create_time / 1000000) as oDate
      from grouper_memberships gmship
      join grouper_members gm on gm.id = gmship.member_id
      join grouper_groups gg on gg.id = gmship.owner_group_id
      join grouper_fields gf on gf.id = gmship.field_id
      where gm.subject_identifier0 = 'blackcm'
      union all
      -- Historical memberships (no display_name in PIT)
      select
        pitm.subject_id,
        pitm.subject_identifier0,
        pitg.name as group_name,
        pitf.name as field_name,
        pitms.active as enabled,
        to_timestamp(pitms.start_time / 1000000) as start_time,
        to_timestamp(pitms.end_time / 1000000) as end_time,
        null as display_name
        ,to_timestamp(GREATEST(EXTRACT(EPOCH FROM to_timestamp(pitms.start_time / 1000000)),
                        EXTRACT(EPOCH FROM to_timestamp(COALESCE(pitms.end_time, 0) / 1000000)))) as oDate
      from grouper_pit_memberships pitms
      join grouper_pit_members pitm on pitm.id = pitms.member_id
      join grouper_pit_groups pitg on pitg.id = pitms.owner_group_id
      join grouper_pit_fields pitf on pitf.id = pitms.field_id
      where pitm.subject_identifier0 = 'blackcm'
) AS union_result   
      order by oDate desc
